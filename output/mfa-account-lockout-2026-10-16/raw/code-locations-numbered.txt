 260: 
 261:         $this->resetAttempts($key);
 262: 
 263:         $this->cache->forget($key.':timer');
 264:     }

 266:     /**
 267:      * Get the number of seconds until the "key" is accessible again.
 268:      *
 269:      * @param  string  $key
 270:      * @return int
 271:      */
 272:     public function availableIn($key)
 273:     {
 274:         $key = $this->cleanRateLimiterKey($key);
 275: 
 276:         return max(0, $this->cache->get($key.':timer') - $this->currentTime());
 277:     }

 279:     /**
 280:      * Clean the rate limiter key from unicode characters.
 281:      *
 282:      * @param  string  $key
 283:      * @return string
 284:      */
 285:     public function cleanRateLimiterKey($key)
 286:     {
 287:         return preg_replace('/&([a-z])[a-z]+;/i', '$1', htmlentities($key));
 288:     }

 290:     /**
 291:      * Execute the given callback without serialization or compression when applicable.
 292:      *
 293:      * @template TReturn
 294:      *
 295:      * @param  (callable(): TReturn)  $callback
 296:      * @return TReturn
 297:      */
 298:     protected function withoutSerializationOrCompression(callable $callback)
 299:     {
 300:         $store = $this->cache->getStore();
 301: 
 302:         if (! $store instanceof RedisStore) {
 303:             return $callback();
 304:         }
 305: 
 306:         $connection = $store->connection();
 307: 
 308:         if (! $connection instanceof PhpRedisConnection) {
 309:             return $callback();
 310:         }
 311: 
 312:         return $connection->withoutSerializationOrCompression($callback);
 313:     }
